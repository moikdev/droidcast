    with tarfile.open(fileobj=io.BytesIO(data)) as archive:
        for member in archive:

                continue

                raise RuntimeError('Cohort precondition or verification failed')
            hashes[member.name] = sha(archive.extractfile(member).read())
    if not hashes == expected['hashes']:
        raise RuntimeError('Existing bytes changed')
    return len(hashes)
