prefixes=[(str(Path.cwd()),'<checkout>'),('<android-sdk>','<android-sdk>'),('<gradle-cache>','<gradle-cache>'),('<user-home>','<user-home>')]
def emit(path,data,original=None,transform=None):
    require(not path.exists(),'Duplicate publication destination')
    path.parent.mkdir(parents=True,exist_ok=True);path.write_bytes(data)
    if original is not None:
        bindings.append(dict(public=str(path),original_relative=str(original.relative_to(base)),original_sha256=sha(original.read_bytes()),public_sha256=sha(data),transform=transform))
def copy(original,path,xml=False,raw=False):
    data=original.read_bytes()
    if raw:transform='raw bytes'
    else:
        for source,target in prefixes:
            if xml:target=target.replace('<','&lt;').replace('>','&gt;')
            data=data.replace(source.encode(),target.encode())
        if xml:data=re.sub(rb'hostname="[^"]*"',b'hostname="redacted-host"',data)
        transform='XML-safe host-prefix/hostname redaction' if xml else 'host-prefix redaction'
    emit(path,data,original,transform)
def json_file(path,value):emit(path,(json.dumps(value,indent=2)+'\n').encode())
def project(original,path,keys):
    data=read(original);value={k:data[k] for k in keys if k in data};json_file(path,value)
    projections.append(dict(public=str(path),original_relative=str(original.relative_to(base)),original_sha256=sha(original.read_bytes()),keys=list(value)))
    return value

slugs={
 'touchchainingreturnstofocusedsearchandkeepslauncherstate':'touch-chain',

def captures(root,cohort,phases):
    observed=read(root/'state.json')
    for original in sorted((root/'cohort/captures').glob('*-frame.json')):
        data=read(original);phase=data['phase']
        if phase not in phases:continue
        stem=original.name.removesuffix('-frame.json')
        method=next((m for m in slugs if stem.startswith('calc-'+m+'-')),None)
        require(method is not None,'Unmapped selected frame')
        name=slugs[method]+'-'+phase
        png=screens/cohort/(name+'.png');xml=public/'captures'/cohort/(name+'.xml');frame=public/'captures'/cohort/(name+'-frame.json')
        copy(original.with_name(stem+'.png'),png,raw=True)
        copy(original.with_name(stem+'.xml'),xml,raw=True)
        copy(original,frame,raw=True)
        frames.append(dict(cohort=cohort,method=method,phase=phase,png=str(png),xml=str(xml),frame=str(frame),app_sha256=observed['app_sha256'],test_apk_sha256=observed['test_apk_sha256'],
            compiled_build_commit=data.get('build_commit'),runtime_source_commit=observed.get('app_source_commit'),
            png_sha256=sha(png.read_bytes()),xml_sha256=sha(xml.read_bytes()),frame_sha256=sha(frame.read_bytes())))
